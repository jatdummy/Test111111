using System.Collections.Generic;

// 격자 지도에서 최단 경로 찾기 (A*)
public class PathFinder
{
    private static readonly int[] Dx =
        { 1, -1, 0, 0, 1, 1, -1, -1 };
    private static readonly int[] Dy =
        { 0, 0, 1, -1, 1, -1, 1, -1 };

    private readonly int _width;
    private readonly int _height;
    private readonly bool[,] _blocked;

    public PathFinder(int width, int height)
    {
        _width = width;
        _height = height;
        _blocked = new bool[width, height];
    }

    public void SetBlocked(int x, int y)
    {
        _blocked[x, y] = true;
    }

    public List<Node> Find(Node start,
        Node goal)
    {
        var open =
            new PriorityQueue<Node, int>();
        var cameFrom =
            new Dictionary<Node, Node>();
        var cost =
            new Dictionary<Node, int>();
        var closed = new HashSet<Node>();

        cost[start] = 0;
        open.Enqueue(start,
            Heuristic(start, goal));

        while (open.Count > 0)
        {
            Node current = open.Dequeue();
            if (current.Equals(goal))
            {
                return Rebuild(cameFrom,
                    current);
            }

            if (!closed.Add(current))
            {
                continue;
            }

            for (int i = 0; i < Dx.Length; i++)
            {
                int nx = current.X + Dx[i];
                int ny = current.Y + Dy[i];
                if (!InBounds(nx, ny)) continue;
                if (_blocked[nx, ny]) continue;

                var next = new Node(nx, ny);
                bool diagonal = i >= 4;
                int step = diagonal ? 14 : 10;
                int newCost =
                    cost[current] + step;

                if (cost.TryGetValue(next,
                    out int old)
                    && old <= newCost)
                {
                    continue;
                }

                cost[next] = newCost;
                cameFrom[next] = current;
                int priority = newCost
                    + Heuristic(next, goal);
                open.Enqueue(next, priority);
            }
        }

        // 경로가 없으면 빈 목록
        return new List<Node>();
    }

    private bool InBounds(int x, int y)
    {
        return x >= 0 && y >= 0
            && x < _width && y < _height;
    }

    private static int Heuristic(Node a,
        Node b)
    {
        int dx = System.Math.Abs(a.X - b.X);
        int dy = System.Math.Abs(a.Y - b.Y);
        int diag = System.Math.Min(dx, dy);
        int straight =
            System.Math.Max(dx, dy) - diag;
        return diag * 14 + straight * 10;
    }

    private static List<Node> Rebuild(
        Dictionary<Node, Node> cameFrom,
        Node current)
    {
        var path = new List<Node>();
        path.Add(current);
        while (cameFrom.TryGetValue(current,
            out Node prev))
        {
            current = prev;
            path.Add(current);
        }

        path.Reverse();
        return path;
    }
}

public readonly struct Node
{
    public readonly int X;
    public readonly int Y;

    public Node(int x, int y)
    {
        X = x;
        Y = y;
    }

    public override int GetHashCode()
    {
        return X * 73856093 ^ Y * 19349663;
    }

    public override bool Equals(object obj)
    {
        return obj is Node n
            && n.X == X && n.Y == Y;
    }
}
